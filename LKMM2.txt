local function getAlivePlayers()
    local alive = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
            table.insert(alive, plr)
        end
    end
    return alive
end

local function getRole(plr)
    local char = plr.Character
    if not char then return "Inocente" end

    if char:GetAttribute("IsMurderer") then
        return "Assassino"
    elseif char:GetAttribute("IsSheriff") then
        return "Sheriff"
    end

    for _, item in ipairs(char:GetChildren()) do
        if item:IsA("Tool") then
            local name = item.Name:lower()
            if name:find("knife") or name:find("murder") or name:find("faca") then
                return "Assassino"
            elseif name:find("gun") or name:find("sheriff") or name:find("arma") then
                return "Sheriff"
            end
        end
    end

    local backpack = plr:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                local name = item.Name:lower()
                if name:find("knife") or name:find("murder") or name:find("faca") then
                    return "Assassino"
                elseif name:find("gun") or name:find("sheriff") or name:find("arma") then
                    return "Sheriff"
                end
            end
        end
    end

    local roleFolder = char:FindFirstChild("Role") or char:FindFirstChild("Murderer") or char:FindFirstChild("Sheriff")
    if roleFolder then
        if roleFolder.Name == "Murderer" or (roleFolder:IsA("BoolValue") and roleFolder.Value and roleFolder.Name == "IsMurderer") then
            return "Assassino"
        elseif roleFolder.Name == "Sheriff" or (roleFolder:IsA("BoolValue") and roleFolder.Value and roleFolder.Name == "IsSheriff") then
            return "Sheriff"
        end
    end

    return "Inocente"
end

local function roleColor(role)
    if role == "Assassino" then return Color3.fromRGB(255,0,0)
    elseif role == "Sheriff" then return Color3.fromRGB(0,100,255)
    else return Color3.fromRGB(0,255,0) end
end

local function hasLineOfSight(targetPosition)
    local camPos = Camera.CFrame.Position
    local ray = Ray.new(camPos, (targetPosition - camPos).Unit * 500)
    local hit = Workspace:FindPartOnRayWithIgnoreList(ray, {player.Character}, false, true)
    return hit == nil
end

local function worldToScreen(position)
    local screenPos, onScreen = Camera:WorldToScreenPoint(position)
    return Vector2.new(screenPos.X, screenPos.Y), onScreen
end

local function getSheriffPlayer()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character and getRole(plr) == "Sheriff" then
            return plr
        end
    end
    return nil
end

local function findSheriffGunOnGround()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Tool") then
            local name = obj.Name:lower()
            if name:find("gun") or name:find("sheriff") or name:find("arma") then
                local parent = obj.Parent
                if parent and not parent:FindFirstChild("Humanoid") then
                    return obj
                end
            end
        end
    end
    return nil
end

local function findSheriffGunInHand()
    local sheriff = getSheriffPlayer()
    if sheriff and sheriff.Character then
        for _, item in ipairs(sheriff.Character:GetChildren()) do
            if item:IsA("Tool") and (item.Name:lower():find("gun") or item.Name:lower():find("sheriff")) then
                return item, sheriff
            end
        end
        local backpack = sheriff:FindFirstChild("Backpack")
        if backpack then
            for _, item in ipairs(backpack:GetChildren()) do
                if item:IsA("Tool") and (item.Name:lower():find("gun") or item.Name:lower():find("sheriff")) then
                    return item, sheriff
                end
            end
        end
    end
    return nil, nil
end

local function getSheriffGun()
    local gunOnGround = findSheriffGunOnGround()
    if gunOnGround then
        return gunOnGround, nil
    end
    local gunInHand, sheriff = findSheriffGunInHand()
    if gunInHand then
        return gunInHand, sheriff
    end
    return nil, nil
end

local function startWeaponESP()
    if weaponESPConnection then weaponESPConnection:Disconnect() end
    weaponESPConnection = RunService.RenderStepped:Connect(function()
        local gun = findSheriffGunOnGround()
        if gun then
            if not gun:FindFirstChild("WeaponHighlight") then
                local highlight = Instance.new("Highlight")
                highlight.Name = "WeaponHighlight"
                highlight.FillColor = Color3.fromRGB(255,128,0)
                highlight.FillTransparency = 0.5
                highlight.OutlineColor = Color3.fromRGB(255,128,0)
                highlight.OutlineTransparency = 0
                highlight.Parent = gun
            end
        else
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("Highlight") and obj.Name == "WeaponHighlight" then
                    obj:Destroy()
                end
            end
        end
    end)
end

local function stopWeaponESP()
    if weaponESPConnection then weaponESPConnection:Disconnect() end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Highlight") and obj.Name == "WeaponHighlight" then
            obj:Destroy()
        end
    end
end

local function startWeaponNotify()
    if weaponNotifyConnection then weaponNotifyConnection:Disconnect() end
    local lastState = false
    weaponNotifyConnection = RunService.Heartbeat:Connect(function()
        local gun = findSheriffGunOnGround()
        local currentState = gun ~= nil
        if currentState and not lastState then
            notify("🔫 Arma do Xerife está no chão!")
        end
        lastState = currentState
    end)
end

local function stopWeaponNotify()
    if weaponNotifyConnection then weaponNotifyConnection:Disconnect() end
end

local function teleportToGun()
    local gun, sheriff = getSheriffGun()
    local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    if gun and not sheriff then
        local pos = gun:IsA("BasePart") and gun.Position or gun:GetPivot().Position
        root.CFrame = CFrame.new(pos + Vector3.new(0,2,0))
        notify("Teleportado para a arma do Xerife (chão)")
    elseif gun and sheriff then
        if sheriff.Character and sheriff.Character:FindFirstChild("HumanoidRootPart") then
            root.CFrame = sheriff.Character.HumanoidRootPart.CFrame + Vector3.new(0,2,0)
            notify("Teleportado para o Xerife (arma com ele)")
        else
            notify("Xerife não encontrado")
        end
    else
        notify("Arma do Xerife não encontrada")
    end
end

local function teleportAllPlayers()
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then
        notify("Seu personagem não está carregado")
        return
    end

    local targetCFrame = myRoot.CFrame * CFrame.new(0, 0, -5)
    local count = 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character then
            local root = plr.Character:FindFirstChild("HumanoidRootPart")
            if root then
                plr.Character:PivotTo(targetCFrame)
                count = count + 1
            end
        end
    end
    notifyShort("Teleportando... (" .. count .. " jogadores)")
end

local function startESPShot()
    stopESPShot()
    espShotActive = true

    espShotStartPart = Instance.new("Part")
    espShotStartPart.Size = Vector3.new(0.1, 0.1, 0.1)
    espShotStartPart.Transparency = 1
    espShotStartPart.CanCollide = false
    espShotStartPart.Anchored = true
    espShotStartPart.Parent = Workspace

    espShotEndPart = Instance.new("Part")
    espShotEndPart.Size = Vector3.new(0.1, 0.1, 0.1)
    espShotEndPart.Transparency = 1
    espShotEndPart.CanCollide = false
    espShotEndPart.Anchored = true
    espShotEndPart.Parent = Workspace

    local beam = Instance.new("Beam")
    beam.Attachment0 = Instance.new("Attachment", espShotStartPart)
    beam.Attachment1 = Instance.new("Attachment", espShotEndPart)
    beam.Color = ColorSequence.new(Color3.fromRGB(255,0,0))
    beam.Width0 = 0.2
    beam.Width1 = 0.2
    beam.FaceCamera = true
    beam.Parent = Workspace
    espShotBeam = beam

    espShotConnection = RunService.RenderStepped:Connect(function()
        local murderer = getMurdererPlayer()
        local myChar = player.Character
        if not murderer or not myChar then
            espShotStartPart.Position = myChar and myChar.HumanoidRootPart.Position or Vector3.new(0,0,0)
            espShotEndPart.Position = espShotStartPart.Position
            return
        end

        local startPos = myChar.HumanoidRootPart.Position + Vector3.new(0,1.5,0)
        local tool = myChar:FindFirstChildOfClass("Tool")
        if tool and tool:FindFirstChild("Handle") then
            startPos = tool.Handle.Position
        end

        local murdererHead = murderer.Character and murderer.Character:FindFirstChild("Head")
        local endPos = murdererHead and murdererHead.Position or startPos

        espShotStartPart.Position = startPos
        espShotEndPart.Position = endPos
    end)

    notify("ESP Shot ativado")
end

local function stopESPShot()
    if espShotConnection then
        espShotConnection:Disconnect()
        espShotConnection = nil
    end
    if espShotBeam then
        espShotBeam:Destroy()
        espShotBeam = nil
    end
    if espShotStartPart then
        espShotStartPart:Destroy()
        espShotStartPart = nil
    end
    if espShotEndPart then
        espShotEndPart:Destroy()
        espShotEndPart = nil
    end
    espShotActive = false
end

local function shootMurder()
    local murderer = getMurdererPlayer()
    if not murderer or not murderer.Character then
        notify("Assassino não encontrado")
        return
    end

    local tool = player.Character and player.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local backpack = player:FindFirstChild("Backpack")
        if backpack then
            for _, item in ipairs(backpack:GetChildren()) do
                if item:IsA("Tool") and (item.Name:lower():find("gun") or item.Name:lower():find("sheriff") or item.Name:lower():find("arma")) then
                    player.Character.Humanoid:EquipTool(item)
                    tool = item
                    break
                end
            end
        end
    end

    if not tool then
        notify("Você não tem uma arma de fogo")
        return
    end

    tool:Activate()
    wait(0.1)
    tool:Deactivate()

    notify("Tiro disparado no assassino")
end

local function startESP()
    if espConnection then espConnection:Disconnect() end
    espConnection = RunService.RenderStepped:Connect(function()
        for _, obj in ipairs(espObjects) do
            if obj then obj:Destroy() end
        end
        espObjects = {}

        for _, plr in ipairs(getAlivePlayers()) do
            local char = plr.Character
            local root = char:FindFirstChild("HumanoidRootPart")
            local head = char:FindFirstChild("Head")
            if root and head then
                local role = getRole(plr)
                local color = roleColor(role)

                if not char:FindFirstChild("ESP_Highlight") then
                    local highlight = Instance.new("Highlight")
                    highlight.Name = "ESP_Highlight"
                    highlight.FillColor = color
                    highlight.FillTransparency = 0.7
                    highlight.OutlineColor = color
                    highlight.OutlineTransparency = 0
                    highlight.Parent = char
                    table.insert(espObjects, highlight)
                end

                local billboard = char:FindFirstChild("ESP_Billboard")
                if not billboard then
                    billboard = Instance.new("BillboardGui")
                    billboard.Name = "ESP_Billboard"
                    billboard.Size = UDim2.new(0,200,0,50)
                    billboard.StudsOffset = Vector3.new(0,2.5,0)
                    billboard.AlwaysOnTop = true
                    billboard.Parent = char

                    local label = Instance.new("TextLabel")
                    label.Size = UDim2.new(1,0,1,0)
                    label.BackgroundTransparency = 1
                    label.TextColor3 = color
                    label.Font = Enum.Font.GothamBold
                    label.TextSize = 14
                    label.Parent = billboard
                    table.insert(espObjects, billboard)
                end
                local label = billboard:FindFirstChildOfClass("TextLabel")
                if label then
                    local distance = (root.Position - player.Character.HumanoidRootPart.Position).Magnitude
                    label.Text = plr.Name .. " [" .. role .. "]\n" .. math.floor(distance) .. "m"
                end
            end
        end
    end)
end

local function startAimbot(mode)
    if aimbotConnection then aimbotConnection:Disconnect() end
    aimbotActive = true
    aimbotMode = mode

    local fovSize = (mode == "Assassino") and 120 or 200

    aimbotConnection = RunService.RenderStepped:Connect(function()
        if not aimbotActive then return end

        local myChar = player.Character
        if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return end

        local enemies = {}
        for _, plr in ipairs(getAlivePlayers()) do
            local role = getRole(plr)
            if mode == "Assassino" then
                if role ~= "Assassino" then
                    table.insert(enemies, plr)
                end
            elseif mode == "Sheriff" then
                if role == "Assassino" then
                    table.insert(enemies, plr)
                end
            end
        end

        local bestTarget = nil
        local bestDistance = math.huge

        for _, plr in ipairs(enemies) do
            local char = plr.Character
            local head = char and char:FindFirstChild("Head")
            if head then
                local screenPos, onScreen = worldToScreen(head.Position)
                if onScreen then
                    local screenCenter = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
                    local distanceFromCenter = (screenPos - screenCenter).Magnitude
                    if distanceFromCenter <= fovSize and hasLineOfSight(head.Position) then
                        if distanceFromCenter < bestDistance then
                            bestDistance = distanceFromCenter
                            bestTarget = head
                        end
                    end
                end
            end
        end

        if bestTarget then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, bestTarget.Position)
        end
    end)

    if showFOV then
        showFOVCircle()
    end
end

local function stopAimbot()
    aimbotActive = false
    if aimbotConnection then aimbotConnection:Disconnect() end
    aimbotConnection = nil
    aimbotMode = nil
    hideFOVCircle()
end

floating.MouseButton1Click:Connect(function()
    main.Visible = not main.Visible
end)

local dragging = false
local dragStart
local startPosition

main.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = main.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
